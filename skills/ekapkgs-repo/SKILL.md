# Creating an ekapkgs-style Repository

This guide covers the files needed to create a new Nix package repository that
integrates with the Ekala ecosystem. The primary input is **ekapkgs**. In rare
cases where the repo is meant to be consumed *by* ekapkgs (a satellite repo),
use **corepkgs** as the input instead.

## Required Files

```
my-repo/
├── flake.nix
├── flake.lock             # Generated by `nix flake update`
├── default.nix            # Flake-free entry point
├── pins.nix               # Reads flake.lock for non-flake usage
├── pkgs-module.nix        # Exposes overlays for downstream
├── top-level.nix          # Aliases and overrides (final: prev:)
├── pkgs/                  # Package expressions (auto-registered)
├── ci/
│   ├── eval.nix           # Evaluates every package
│   ├── eval.sh            # Shell wrapper for CI
│   └── packages.nix       # Enumerates packages, probe logic
├── .github/
│   └── workflows/
│       └── nix.yml        # Lint + eval jobs
├── .gitignore
└── LICENSE
```

## flake.nix

Most repos take ekapkgs as their single input. Follow the `systems` input from
ekapkgs for `forAllSystems`.

```nix
{
  description = "My package set";

  inputs = {
    ekapkgs.url = "github:ekala-project/ekapkgs";
    systems.follows = "ekapkgs/corepkgs/systems";
  };

  outputs =
    { ekapkgs, self, systems, ... }:
    let
      forAllSystems = ekapkgs.lib.genAttrs (import systems);
    in
    {
      legacyPackages = forAllSystems (
        system:
        import ./. {
          inherit system;
          modules = [ (import ./pkgs-module.nix) ];
        }
      );
    };
}
```

### Satellite repo consumed by ekapkgs

If the repo will be imported *into* ekapkgs as a non-flake input (like
`python-pkgs`, `haskell-pkgs`, etc.), use **corepkgs** directly:

```nix
{
  description = "Satellite package set for ekapkgs";

  inputs = {
    corepkgs.url = "github:ekala-project/corepkgs";
    systems.follows = "corepkgs/systems";
  };

  outputs =
    { corepkgs, self, systems, ... }:
    let
      forAllSystems = corepkgs.lib.genAttrs (import systems);
    in
    {
      legacyPackages = forAllSystems (
        system:
        import ./. {
          inherit system;
          modules = [ (import ./pkgs-module.nix) ];
        }
      );
    };
}
```

Satellite repos set `flake = false` in the consuming flake so they are imported
as plain source trees via `pins.nix`, not as flake outputs.

## pins.nix

Reads `flake.lock` so the repo can be imported without the flake evaluator.
List every input you need.

```nix
let
  lock = builtins.fromJSON (builtins.readFile ./flake.lock);
  getInfo =
    attr:
    let
      node = lock.nodes.${attr}.locked;
    in
    if node.type == "path" then
      builtins.fetchTree {
        inherit (node) type path narHash;
      }
    else
      builtins.fetchTree {
        inherit (node)
          type
          owner
          repo
          narHash
          rev
          ;
      };
in
{
  ekapkgs = getInfo "ekapkgs";
}
```

For satellite repos, replace `ekapkgs` with `corepkgs`:

```nix
{
  corepkgs = getInfo "corepkgs";
}
```

## default.nix

Flake-free entry point. Imports the upstream package set and applies local
modules.

```nix
let
  pins = import ./pins.nix;
  upstream = import pins.ekapkgs;
  pkgsModule = import ./pkgs-module.nix;
in

{
  modules ? [ ],
  ...
}@args:

let
  filteredAttrs = builtins.removeAttrs args [ "modules" ];
in

upstream (
  {
    modules = modules ++ [ pkgsModule ];
  }
  // filteredAttrs
)
```

For satellite repos importing corepkgs:

```nix
let
  pins = import ./pins.nix;
  upstream = import pins.corepkgs;
  pkgsModule = import ./pkgs-module.nix;
in
# ...same pattern
```

## pkgs-module.nix

Exposes overlays for downstream consumption. Uses
`lib.packageSets.mkAutoCalledPackageDir` to auto-register packages from the
`pkgs/` directory.

```nix
{ lib, config, ... }:
let
  pkgsOverlay = lib.packageSets.mkAutoCalledPackageDir ./pkgs;
  pkgsOverrides = import ./top-level.nix;
in
{
  overlays.pkgs = [
    pkgsOverlay
    pkgsOverrides
  ];
}
```

## top-level.nix

Overlay for aliases, variant wrappers, and overrides that cannot be expressed as
a plain `callPackage`. Starts minimal.

```nix
final: prev: {
  # Aliases and overrides go here
}
```

## ci/packages.nix

Enumerates every package for evaluation. Disables aliases to avoid redundant
work. Provides a `probe` function that forces `drvPath` to catch real errors.

```nix
{
  checkMeta ? false,
}:

let
  pkgs = import ../. {
    config = {
      inherit checkMeta;
      aliases.nixpkgs = false;

      handleEvalIssue =
        reason: msg:
        if
          builtins.elem reason [
            "unknown-meta"
            "broken-outputs"
          ]
        then
          abort msg
        else
          throw msg;
    };
  };

  inherit (pkgs) lib;

  probe =
    package:
    let
      value = builtins.tryEval package;
    in
    builtins.tryEval (
      if value.success && lib.isDerivation value.value then
        builtins.seq value.value.drvPath null
      else
        null
    );

  isBuildable =
    value:
    let
      result = builtins.tryEval (lib.isDerivation value && builtins.seq value.drvPath true);
    in
    result.success && result.value;

  names = builtins.attrNames pkgs;

  topLevel = map (name: {
    inherit name;
    value = pkgs.${name};
  }) names;

  targets = map (pair: pair.value) topLevel;
in
{
  inherit probe targets;
  buildable = builtins.listToAttrs (builtins.filter (pair: isBuildable pair.value) topLevel);
}
```

## ci/eval.nix

Drives the evaluation. Forces every package and reports the count.

```nix
let
  packages = import ./packages.nix { };
  inherit (packages) probe targets;
in
builtins.deepSeq (map probe targets) {
  evaluated = builtins.length targets;
}
```

## ci/eval.sh

Shell wrapper for CI. Fails on evaluation errors and Nix warnings.

```bash
#!/usr/bin/env bash
set -euo pipefail

cd "$(dirname "$0")/.."

stderr="$(mktemp)"
trap 'rm -f "$stderr"' EXIT

if ! result="$(nix-instantiate --eval --strict --json ci/eval.nix --show-trace 2>"$stderr")"; then
  cat "$stderr" >&2
  exit 1
fi

if grep -q '^warning:' "$stderr"; then
  cat "$stderr" >&2
  echo
  echo "Evaluation emitted the warnings above; fix them or the job stays red."
  exit 1
fi

count="$(sed -E 's/.*"evaluated":([0-9]+).*/\1/' <<<"$result")"

echo "All $count packages evaluate."
```

Make it executable: `chmod +x ci/eval.sh`

## .github/workflows/nix.yml

Two jobs: lint (format check) and eval (evaluate all packages).

```yaml
name: Nix

on:
  push:
    branches: [ master ]
  pull_request:
    branches: [ master ]

jobs:
  lint:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v4
    - uses: cachix/install-nix-action@v30
      with:
        extra_nix_config: |
          extra-substituters = https://ekala-corepkgs.cachix.org
          extra-trusted-public-keys = ekala-corepkgs.cachix.org-1:DcZV+vegWoEzacbSdXFXU4S7728C0eS9RfGpKeyHd6w=
    - name: Nix Lint
      run: nix fmt . -- --ci

  eval:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v4
    - uses: cachix/install-nix-action@v30
      with:
        extra_nix_config: |
          extra-substituters = https://ekala-corepkgs.cachix.org
          extra-trusted-public-keys = ekala-corepkgs.cachix.org-1:DcZV+vegWoEzacbSdXFXU4S7728C0eS9RfGpKeyHd6w=
    - name: Evaluate all packages
      run: ./ci/eval.sh
```

## Checklist

- [ ] `flake.nix` with ekapkgs (or corepkgs) as input
- [ ] `pins.nix` mirrors flake inputs
- [ ] `default.nix` imports upstream and applies `pkgs-module.nix`
- [ ] `pkgs-module.nix` registers `pkgs/` via `mkAutoCalledPackageDir`
- [ ] `top-level.nix` exists (can be empty overlay)
- [ ] `ci/eval.sh`, `ci/eval.nix`, `ci/packages.nix` in place
- [ ] `.github/workflows/nix.yml` with lint + eval jobs
- [ ] `nix flake update` to generate `flake.lock`
- [ ] `./ci/eval.sh` passes locally
